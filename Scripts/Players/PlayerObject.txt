private table Player_SonicPhysicsTable
	0x60000, 0x0C00, 0x1800, 0x0600, 0x08000, 0x0600, -0x68000, -0x40000
	0x30000, 0x0600, 0x0C00, 0x0300, 0x04000, 0x0300, -0x38000, -0x20000
end table
private table Player_TailsPhysicsTable
	0x60000, 0x0C00, 0x1800, 0x0600, 0x08000, 0x0600, -0x68000, -0x40000
	0x30000, 0x0600, 0x0C00, 0x0300, 0x04000, 0x0300, -0x38000, -0x20000
end table
private table Player_KnucklesPhysicsTable
	0x60000, 0x0C00, 0x1800, 0x0600, 0x08000, 0x0600, -0x60000, -0x40000
	0x30000, 0x0600, 0x0C00, 0x0300, 0x04000, 0x0300, -0x30000, -0x20000
end table

public function Player_UpdatePhysicsState
    temp0 = 0
    switch object.propertyValue
        case 0
        case 1
            temp0 = 0
            break
        case 2
            temp0 = 1
            break
        case 3
            temp0 = 2
            break
    end switch
    temp1 = 0
    if object.ypos > stage.waterLevel
        temp1 = 8
        object.value24 = 4096
    else
        object.value24 = 14336
    end if
    switch temp0
        case 0
            GetTableValue(object.value19, temp1, Player_SonicPhysicsTable)
            temp1++
            GetTableValue(object.value20, temp1, Player_SonicPhysicsTable)
            temp1 += 5
            GetTableValue(object.value26, temp1, Player_SonicPhysicsTable)
            temp1++
            GetTableValue(object.value27, temp1, Player_SonicPhysicsTable)
            break
        case 1
            GetTableValue(object.value19, temp1, Player_TailsPhysicsTable)
            temp1++
            GetTableValue(object.value20, temp1, Player_TailsPhysicsTable)
            temp1 += 5
            GetTableValue(object.value26, temp1, Player_TailsPhysicsTable)
            temp1++
            GetTableValue(object.value27, temp1, Player_TailsPhysicsTable)
            break
        case 2
            GetTableValue(object.value19, temp1, Player_KnucklesPhysicsTable)
            temp1++
            GetTableValue(object.value20, temp1, Player_KnucklesPhysicsTable)
            temp1 += 5
            GetTableValue(object.value26, temp1, Player_KnucklesPhysicsTable)
            temp1++
            GetTableValue(object.value27, temp1, Player_KnucklesPhysicsTable)
            break
    end switch
    object.value21 = object.value20
end function

public function Player_ProcessUpdate
    if object.value15 == true
        if object.value14 == true
            object.up = keyDown[0].up
            object.down = keyDown[0].down
            object.left = keyDown[0].left
            object.right = keyDown[0].right
            object.jumpHold = keyDown[0].buttonA
            object.jumpHold |= keyDown[0].buttonB
            object.jumpHold |= keyDown[0].buttonC
            object.jumpPress = keyPress[0].buttonA
            object.jumpPress |= keyPress[0].buttonB
            object.jumpPress |= keyPress[0].buttonC
        else
            temp0 = saveRAM[1200]
            temp0 -= 32
            temp0 &= 63
            temp1 = 1000 + temp0
            temp3 = saveRAM[temp1]
            temp1 = 1100 + temp0
            temp4 = saveRAM[temp1]
            object.left = false
            object.right = false
            object.jumpPress = false
            if object.xpos < temp3
                object.right = true
            else
                if object.xpos > temp3
                    object.left = true
                end if
            end if
            if object.gravity == 0
                if object.ypos > temp4
                    object.jumpPress = true
                end if
            else
                if object.ypos > temp4
                    object.jumpHold = true
                else
                    object.jumpHold = false
                end if
            end if
        end if
    else
        ProcessObjectControl()
        if playerCount > 1
            if object.state == Player_State_Air
                if object[1].state == Player_State_Fly
                    if object[1].value14 == false
                        temp0 = object.xpos
                        temp0 -= object[1].xpos
                        Abs(temp0)
                        if temp0 < 0x100000
                            temp0 = object.ypos
                            temp0 -= object[1].ypos
                            Abs(temp0)
                            if temp0 < 0x180000
                                object.state = Player_State_Carried
                                object[1].value14 = true
                            end if
                        end if
                    end if
                end if
            end if
        end if
        if object.entityPos == 0
            temp0 = saveRAM[1200]
            temp1 = 1000 + temp0
            saveRAM[temp1] = object.xpos
            temp1 = 1100 + temp0
            saveRAM[temp1] = object.ypos
            temp0++
            temp0 &= 63
            saveRAM[1200] = temp0
        end if
    end if
end function

public function Player_Action_DropDash
    if object.propertyValue < 2
        if object.animation == 2
            if object.jumpHold == true
                object.value2++
                if object.value2 >= 20
                    object.animation = 48
                    if object.value2 == 20
                        PlaySfx(SfxName[Charge], 0)
                    end if
                end if
            else
                object.value2 = 0
            end if
        end if
    end if
end function

public function Player_Action_Glide
    if object.jumpPress == true
        object.state = Player_State_Glide
        object.animation = 44
        object.yvel = 0x8000
        if object.direction == 0
            object.speed = 0x40000
        else
            object.speed = -0x40000
        end if
        object.xvel = object.speed
    end if
end function

public function Player_State_Glide
    object.yvel = 0x8000
    object.xvel = object.speed
    if object.jumpHold == false
        object.state = Player_State_GlideDrop
        object.animation = 45
    end if
    temp0 = 10
    if object.direction != 0
        temp0 = -10
    end if
    ObjectTileCollision(2, temp0, 0, object.value33)
    if checkResult == true
        object.state = Player_State_Climb
        object.animation = 46
        object.speed = 0
        object.xvel = 0
    end if
    if object.gravity == 0
        object.state = Player_State_Ground
        object.animation = 0
    end if
end function

public function Player_State_GlideDrop
    object.yvel += object.value24
    object.xvel = 0
    if object.gravity == 0
        object.state = Player_State_Ground
        object.animation = 0
    end if
end function

public function Player_State_Climb
    object.xvel = 0
    object.yvel = 0
    if object.up == true
        object.ypos -= 0x10000
    else
        if object.down == true
            object.ypos += 0x10000
        end if
    end if
    temp0 = 10
    if object.direction != 0
        temp0 = -10
    end if
    ObjectTileCollision(2, temp0, -16, object.value33)
    if checkResult == false
        object.state = Player_State_LedgePullUp
        object.animation = 47
        object.value1 = 0
    end if
    if object.jumpPress == true
        object.state = Player_State_Air
        object.yvel = -0x40000
    end if
end function

public function Player_State_LedgePullUp
    object.value1++
    if object.value1 > 20
        object.state = Player_State_Ground
        object.ypos -= 0x180000
        if object.direction == 0
            object.xpos += 0x100000
        else
            object.xpos -= 0x100000
        end if
    end if
end function

public function Player_State_Carried
    object.xpos = object[0].xpos
    object.ypos = object[0].ypos
    object.ypos += 0x100000
    object.direction = object[0].direction
    if object.jumpPress == true
        object.state = Player_State_Air
        object.yvel = -0x40000
        object[0].value14 = false
    end if
    if object[0].state != Player_State_Fly
        object.state = Player_State_Air
        object[0].value14 = false
    end if
end function

public function Player_State_Ground
    ProcessObjectMovement()
end function

public function Player_State_Air
    ProcessObjectMovement()
end function

public function Player_State_Fly
    ProcessObjectMovement()
end function

event ObjectUpdate
    CallFunction(Player_ProcessUpdate)
    CallFunction(object.state)
    ProcessAnimation()
    if object.state == Player_State_Air
        if object.propertyValue < 2
            CallFunction(Player_Action_DropDash)
        else
            if object.propertyValue == 3
                CallFunction(Player_Action_Glide)
            end if
        end if
    end if
end event

event ObjectStartup
    switch object[0].propertyValue
        case 0
            LoadAnimation("Sonic.ani")
            object[0].value29 = -5
            ResetObjectEntity(1, TypeName[Tails Object], 2, object[0].xpos, object[0].ypos)
            object[1].value15 = true
            playerCount = 2
            break
        case 1
            LoadAnimation("Sonic.ani")
            object[0].value29 = -5
            playerCount = 1
            break
        case 2
            LoadAnimation("Tails.ani")
            object[0].value29 = -1
            playerCount = 1
            break
        case 3
            LoadAnimation("Knuckles.ani")
            object[0].value29 = -5
            playerCount = 1
            break
    end switch
    object.state = Player_State_Air
    CallFunction(Player_UpdatePhysicsState)
    if playerCount > 1
        // We can't easily CallFunction on another object, but they will update themselves
    end if
end event

event ObjectDraw
    DrawObjectAnimation()
end event
