// Initialize the character object for replay playback
public function Player_SetupAttractDemo
	Player_attractTablePos = 2
	Player_attractFrameCount = 1

	currentPlayer = 0
	while currentPlayer < playerCount
		GetTableValue(object[currentPlayer].xpos, 0, Player_attractTable)
		GetTableValue(object[currentPlayer].ypos, 1, Player_attractTable)
		object[currentPlayer].controlMode 	= -1 // none
		object[currentPlayer].up 			= false
		object[currentPlayer].down 			= false
		object[currentPlayer].left 			= false
		object[currentPlayer].right 		= false
		object[currentPlayer].jumpPress 	= false
		object[currentPlayer].jumpHold 		= false
		object[currentPlayer].value1 		= 0
		currentPlayer++
	loop
	screen[0].cameraX = object[0].ixpos
	screen[0].cameraY = object[0].iypos
end function


public function Player_ApplyShield
	switch object[currentPlayer].value38
	case SHIELD_NONE
		ResetObjectEntity(arrayPos0, TypeName[Blank Object], 0, 0, 0)
		break
	case SHIELD_NORMAL
		ResetObjectEntity(arrayPos0, blueShieldType, 0, 0, 0)
		object[arrayPos0].priority = PRIORITY_ACTIVE
		object[arrayPos0].inkEffect = INK_ALPHA
		object[arrayPos0].alpha = 160
		break
	case SHIELD_BUBBLE
		ResetObjectEntity(arrayPos0, TypeName[Bubble Shield], 0, 0, 0)
		object[arrayPos0].priority = PRIORITY_ACTIVE
		break
	case SHIELD_FIRE
		ResetObjectEntity(arrayPos0, TypeName[Fire Shield], 0, 0, 0)
		object[arrayPos0].priority = PRIORITY_ACTIVE
		break
	case SHIELD_LIGHTNING
		ResetObjectEntity(arrayPos0, TypeName[Lightning Shield], 0, 0, 0)
		object[arrayPos0].priority = PRIORITY_ACTIVE
		break
	// Unfortunately, SHIELD_GOGGLES doesn't exist here :(
	end switch
end function

// Used by badniks to check if the badnik should be destroyed
// -> Not to be confused with Player_CheckHit
public function Player_BadnikBreak
	CheckEqual(object[currentPlayer].animation, ANI_JUMPING)
	temp0 = checkResult
	CheckEqual(object[currentPlayer].animation, ANI_SPINDASH)
	temp0 |= checkResult
	CheckNotEqual(object[currentPlayer].value7, 0)
	temp0 |= checkResult
	if warp.value1 > 0
		temp0 |= 1
	end if
	CheckEqual(object[currentPlayer].animation, ANI_FLYING)
	temp1 = checkResult
	CheckEqual(object[currentPlayer].animation, ANI_FLYINGTIRED)
	temp1 |= checkResult
	if temp1 == true
		CheckGreater(object[currentPlayer].ypos, player.ypos)
		temp0 |= checkResult
	end if
	if temp0 == true
		ResetObjectEntity(player.entityPos, flowerType, 0, player.xpos, player.ypos)
		player.drawOrder = 4
		CreateTempObject(TypeName[Smoke Puff], 0, player.xpos, player.ypos)
		object[tempObjectPos].drawOrder = 4
		CreateTempObject(TypeName[Object Score], object[currentPlayer].value18, player.xpos, player.ypos)
		object[tempObjectPos].drawOrder = 4
		PlaySfx(SfxName[Destroy], 0)
		
		if object[currentPlayer].yvel > 0
			if object[currentPlayer].ypos >= player.ypos
				object[currentPlayer].yvel -= 65536
			else
				object[currentPlayer].yvel += object[currentPlayer].value24
				object[currentPlayer].yvel += object[currentPlayer].value24
				FlipSign(object[currentPlayer].yvel)
			end if
		else
			object[currentPlayer].yvel += 65536
		end if
		temp0 = currentPlayer
		currentPlayer = 0
		switch object[currentPlayer].value18
		case 0
			player.score += 100
			break
		case 1
			player.score += 200
			break
		case 2
			player.score += 500
			break
		case 3
		case 4
		case 5
		case 6
		case 7
		case 8
		case 9
		case 10
		case 11
		case 12
		case 13
		case 14
			player.score += 1000
			break
		case 15
			player.score += 10000
			break
		end switch
		if object[currentPlayer].value18 < 15
			object[currentPlayer].value18++
		end if
		currentPlayer = temp0
	else
		if object[currentPlayer].state != Player_State_Death
			if object[currentPlayer].value7 == 0
				if object[currentPlayer].value8 == 0
					object[currentPlayer].state = Player_State_GotHit
					if object[currentPlayer].xpos > player.xpos
						object[currentPlayer].speed = 0x20000
					else
						object[currentPlayer].speed = -0x20000
					end if
				end if
			end if
		end if
	end if
end function

public function Player_Kill
	PlaySfx(SfxName[Hurt], 0)

	object[currentPlayer].speed 			= 0
	object[currentPlayer].xvel 				= 0
	object[currentPlayer].yvel 				= -425984
	object[currentPlayer].state 			= Player_State_Death
	object[currentPlayer].animation 		= ANI_DYING
	object[currentPlayer].tileCollisions 	= false
	object[currentPlayer].interaction 		= false
	object[currentPlayer].value8 		= 0
	object[currentPlayer].visible 			= true
	object[currentPlayer].drawOrder 		= 6

	if currentPlayer == 0
		object[currentPlayer].priority = PRIORITY_ALWAYS
		if object[1].type == TypeName[Blank Object] // 0
			object[1].priority = PRIORITY_ALWAYS
		end if

		// Stop the game flow
		screen[0].cameraEnabled = 0
		stage.state = STAGE_FROZEN
	end if

	arrayPos0 = currentPlayer
	arrayPos0 += playerCount

	if object[arrayPos0].type == invincibilityType
		object[arrayPos0].propertyValue = 3
	end if
	
	object[arrayPos0].type = TypeName[Blank Object]
	object[currentPlayer].value38 = SHIELD_NONE
end function


public function Player_ProcessUpdate
	if object.value15 == true
		CallFunction(Player_Sidekick_UpdateAI)
	else
		ProcessObjectControl()
		if playerCount > 1
			if object.state == Player_State_Air
				if object[1].state == Player_State_Fly
					if object[1].value14 == false
						temp0 = object.xpos
						temp0 -= object[1].xpos
						Abs(temp0)
						if temp0 < 1048576
							temp0 = object.ypos
							temp0 -= object[1].ypos
							Abs(temp0)
							if temp0 < 1572864
								object.state = Player_State_Carried
								object[1].value14 = true
							end if
						end if
					end if
				end if
			end if
		end if
		if object.entityPos == 0
			temp0 = saveRAM[1200]
			temp1 = 1000
			temp1 += temp0
			saveRAM[temp1] = object.xpos
			temp1 = 1100
			temp1 += temp0
			saveRAM[temp1] = object.ypos
			temp0++
			temp0 &= 63
			saveRAM[1200] = temp0
		end if
	end if
end function

		Player.State = Player_State_Air

		Player.Animation = ANI_WALKING
		Player.Timer	 = 0
		CallFunction(Player_HandleAirMovement)
	else
		CallFunction(Player_HandleOnGround)
		// Yes, the only thing the function does is make you wait
		if Player.Timer < 240
			Player.Animation = ANI_STOPPED
			Player.Timer++
		else
			Player.Animation = ANI_WAITING
		end if

		// Warp cancel check
		if Warp.Destination > WARPDEST_NONE
			if Warp.Timer > 99
				if Warp.Timer < 204
					Warp.Destination = WARPDEST_NONE
				end if
			end if
			Warp.Timer = 0
		end if

		// Get free from Amy
		if Player.JumpPress == true
			CallFunction(Player_Action_Jump)
		end if
	end if
end function

// NOTE: This state was transplanted from the v3 scripts due to it not existing when decompiling this script.
public function Player_State_Ramp3D
	CallFunction(Player_HandleGroundMovement)
	if player.gravity == GRAVITY_AIR

		player.state = Player_State_Air

		player.value1 = 0
		CallFunction(Player_HandleAirMovement)
		// Set the equivalent animation for looking to the left
		if player.direction == FACING_LEFT

			if player.animation == ANI_RAMP_RUNNING3
				player.animation = ANI_RAMP_RUNNING5
			end if

			if player.animation == ANI_RAMP_RUNNING2
				player.animation = ANI_RAMP_RUNNING6
			end if

		end if
	else
		CallFunction(Player_HandleOnGround)
		if player.speed == 0
			// Warp Star erase check
			if warp.destination > WARPDEST_NONE
				if warp.value1 > 99
					if warp.value1 < 204
						warp.destination = WARPDEST_NONE
					end if
				end if
				warp.value1 = 0
			end if
			// Waiting timer while in the ramp
			if player.value1 < 240
				player.animation = ANI_STOPPED
				player.value1++
			else
				player.animation = ANI_WAITING
			end if
		else
			// Set animation based on angle
			if player.angle == 0
				player.animation = ANI_RAMP_RUNNING1
			end if

			if player.angle > 200
				if player.direction == FACING_RIGHT
					player.animation = ANI_RAMP_RUNNING3
				else
					player.animation = ANI_RAMP_RUNNING5
				end if
			end if

			if player.angle > 216
				if player.direction == FACING_RIGHT
					player.animation = ANI_RAMP_RUNNING2
				else
					player.animation = ANI_RAMP_RUNNING6
				end if
			end if

			if player.angle > 232
				player.animation = ANI_RAMP_RUNNING1
			end if

			if player.angle == 192
				player.animation = ANI_RAMP_RUNNING4
			end if
		end if

		if player.jumpPress == true
			CallFunction(Player_Action_Jump)
		else
			if player.up == true
				if player.speed == 0
					player.state	 = Player_State_LookUp
					player.animation = ANI_LOOKINGUP
					player.value1	 = 0
				end if
			end if
			if player.down == true
				if player.speed == 0
					player.state	 = Player_State_Crouch
					player.animation = ANI_LOOKINGDOWN
					player.value1	 = 0
				else
					if player.speed >  0x199A
						player.controlLock	= 0
						player.state		= Player_State_Roll
						player.animation	= ANI_JUMPING
					end if

					if player.speed < -0x199A
						player.controlLock	= 0
						player.state		= Player_State_Roll
						player.animation	= ANI_JUMPING
					end if
				end if
			end if
		end if
	end if

	if player.animation != ANI_STOPPED
		player.animationSpeed = player.speed
		if player.animationSpeed < 0
			FlipSign(player.animationSpeed)
		end if
		player.animationSpeed *= 60
		player.animationSpeed /= 0x60000
		player.animationSpeed += 20
	else
		player.animationSpeed = 0
	end if
end function

public function Player_HandleGroundMovement
	if player.controlLock > 0
		player.controlLock--
		Sin256(temp0, player.angle)
		temp0 *= 0x2000
		temp0 >>= 8
		player.speed += temp0
	else
		if player.left == true
			temp0 = player.value19
			FlipSign(temp0)
			if player.speed > temp0
				if player.speed > 0
					if player.collisionMode == 0
						if player.speed > 0x40000
							player.skidding = 16
						end if
					end if
					if player.speed < player.skidSpeed
						player.speed = player.skidSpeed
						FlipSign(player.speed)
						player.skidding = 0
					else
						player.speed -= player.skidSpeed
					end if
				else
					player.speed -= player.value20
					player.skidding = 0
				end if
			end if
			if player.speed <= 0
				player.direction = FACING_LEFT
			end if
		end if
		if player.right == true
			if player.speed < player.value19
				if player.speed < 0
					if player.collisionMode == 0
						if player.speed < -0x40000
							player.skidding = 16
						end if
					end if
					temp0 = player.skidSpeed
					FlipSign(temp0)
					if player.speed > temp0
						player.speed = player.skidSpeed
						player.skidding = 0
					else
						player.speed += player.skidSpeed
					end if
				else
					player.speed += player.value20
					player.skidding = 0
				end if
			end if
			if player.speed >= 0
				player.direction = FACING_RIGHT
			end if
		end if
		temp0 = player.left
		temp0 |= player.right
		if temp0 == 0
			if player.speed > 0
				player.speed -= player.value21
				if player.speed < 0
					player.speed = 0
				end if
			else
				player.speed += player.value21
				if player.speed > 0
					player.speed = 0
				end if
			end if
			if player.speed > 0x2000
				Sin256(temp0, player.angle)
				temp0 *= 0x2000
				temp0 >>= 8
				player.speed += temp0
			end if
			if player.speed < -0x2000
				Sin256(temp0, player.angle)
				temp0 *= 0x2000
				temp0 >>= 8
				player.speed += temp0
			end if
			if player.angle > 192
				if player.angle < 228
					if player.speed > -65536
						if player.speed < 65536
							player.controlLock = 30
						end if
					end if
				end if
			end if
			if player.angle > 28
				if player.angle < 64
					if player.speed > -65536
						if player.speed < 65536
							player.controlLock = 30
						end if
					end if
				end if
			end if
		else
			Sin256(temp0, player.angle)
			temp0 *= 0x2000
			temp0 >>= 8
			player.speed += temp0
			if player.right == 1
				if player.left == 0
					if player.angle > 192
						if player.angle < 228
							if player.speed < 163840
								if player.speed > -0x20000
									player.controlLock = 30
								end if
							end if
						end if
					end if
				end if
			else
				if player.left == 1
					if player.angle > 28
						if player.angle < 64
							if player.speed > -163840
								if player.speed < 0x20000
									player.controlLock = 30
								end if
							end if
						end if
					end if
				end if
			end if
		end if
	end if
	switch player.collisionMode
	case 1
		if player.angle <= 192
			if player.speed > -0x20000
				if player.speed < 0x20000
					player.gravity = GRAVITY_AIR
					player.angle = 0
					player.collisionMode = 0
					player.speed = player.xvel
				end if
			end if
		end if
		break
	case 2
		if player.speed > -0x20000
			if player.speed < 0x20000
				player.gravity = GRAVITY_AIR
				player.angle = 0
				player.collisionMode = 0
				player.speed = player.xvel
			end if
		end if
		break
	case 3
		if player.angle >= 64
			if player.speed > -0x20000
				if player.speed < 0x20000
					player.gravity = GRAVITY_AIR
					player.angle = 0
					player.collisionMode = 0
					player.speed = player.xvel
				end if
			end if
		end if
		break
	end switch
end function


public function Player_HandleAirFriction
	if player.yvel > -262144
		if player.yvel < 0
			temp0 = player.speed
			temp0 >>= 5
			player.speed -= temp0
		end if
	end if
	temp0 = player.value19
	FlipSign(temp0)
	if player.speed > temp0
		if player.left == 1
			player.speed -= player.airAcceleration
			player.direction = FACING_LEFT
		end if
	else
		if player.left == 1
			player.direction = FACING_LEFT
		end if
	end if
	if player.speed < player.value19
		if player.right == 1
			player.speed += player.airAcceleration
			player.direction = FACING_RIGHT
		end if
	else
		if player.right == 1
			player.direction = FACING_RIGHT
		end if
	end if
	if options.originalControls == 1
		if player.left == 1
			temp0 = player.value19
			FlipSign(temp0)
			if player.speed < temp0
				player.speed = temp0
			end if
		end if
		if player.right == 1
			if player.speed > player.value19
				player.speed = player.value19
			end if
		end if
	end if
end function


public function Player_HandleRollDeceleration
	if player.right == 1
		if player.speed < 0
			player.speed += player.rollingDeceleration
		end if
	end if
	if player.left == 1
		if player.speed > 0
			player.speed -= player.rollingDeceleration
		end if
	end if
	temp1 = player.speed
	if player.speed > 0
		player.speed -= player.value28
		Sin256(temp0, player.angle)
		if temp0 > 0
			Sin256(temp0, player.angle)
			temp0 *= 20480
		else
			Sin256(temp0, player.angle)
			temp0 *= 5120
		end if
		temp0 >>= 8
		player.speed += temp0
		if player.speed > 1179648
			player.speed = 1179648
		end if
	else
		player.speed += player.value28
		Sin256(temp0, player.angle)
		if temp0 < 0
			Sin256(temp0, player.angle)
			temp0 *= 20480
		else
			Sin256(temp0, player.angle)
			temp0 *= 5120
		end if
		temp0 >>= 8
		player.speed += temp0
		if player.speed < -1179648
			player.speed = -1179648
		end if
	end if
	switch player.collisionMode
	case 0
	case 2
		if temp1 > 0
			if player.speed < 0
				player.speed = 0
				player.state = Player_State_Ground
			end if
		else
			if player.speed > 0
				player.speed = 0
				player.state = Player_State_Ground
			end if
		end if
		break
	case 1
		if player.angle < 193
			if temp1 > 0
				if player.speed < 0x20000
					player.gravity = GRAVITY_AIR
					player.xvel = 0
					player.speed = 0
				end if
			end if
		end if
		break
	case 3
		if player.angle > 63
			if temp1 < 0
				if player.speed > -0x20000
					player.gravity = GRAVITY_AIR
					player.xvel = 0
					player.speed = 0
				end if
			end if
		end if
		break
	end switch
end function


public function Player_HandleAirMovement
	player.scrollTracking = true
	player.yvel += player.value24
	if player.yvel > 1048576
		player.yvel = 1048576
	end if
	if player.yvel < player.value27
		if player.jumpHold == 0
			if player.value1 > 0
				player.yvel = player.value27
				temp0 = player.speed
				temp0 >>= 5
				player.speed -= temp0
			end if
		end if
	end if
	player.xvel = player.speed
	if player.rotation < 256
		if player.rotation > 0
			player.rotation -= 4
		else
			player.rotation = 0
		end if
	else
		if player.rotation < 512
			player.rotation += 4
		else
			player.rotation = 0
		end if
	end if
	player.collisionMode = 0
	if player.animation == ANI_JUMPING
		player.animationSpeed = player.rollAnimationSpeed
	end if
end function


public function Player_HandleOnGround
	player.scrollTracking = 0
	Cos256(temp0, player.angle)
	temp0 *= player.speed
	temp0 >>= 8
	player.xvel = temp0
	Sin256(temp0, player.angle)
	temp0 *= player.speed
	temp0 >>= 8
	player.yvel = temp0
end function


public function Player_Action_Jump
	temp1 = false

	if player.collisionMode == 0
		// Check if there's a roof right above the player
		temp6 = player.xpos
		temp7 = player.ypos
		temp0 = player.collisionTop
		temp0 -= 2

		ObjectTileCollision(3, 0, temp0, player.collisionPlane)
		temp1 = checkResult
		player.xpos = temp6
		player.ypos = temp7
		temp0 = object.collisionBottom
		if player.animation != ANI_JUMPING
			player.iypos -= player.value29
			temp0 += player.value29
		end if
		ObjectTileCollision(0, 0, temp0, player.collisionPlane)
	end if
	if temp1 == false
		// No roof - start jumping
		player.controlLock = 0
		player.gravity = GRAVITY_AIR
		temp1 = player.value26
		temp1 += player.value24
		Sin256(player.xvel, player.angle)
		player.xvel *= temp1
		Cos256(temp0, player.angle)
		temp0 *= player.speed
		player.xvel += temp0
		player.xvel >>= 8
		Sin256(player.yvel, player.angle)
		player.yvel *= player.speed
		Cos256(temp0, player.angle)
		temp0 *= temp1
		player.yvel -= temp0
		player.yvel >>= 8
		player.speed = player.xvel
		player.scrollTracking = 1
		player.animation = ANI_JUMPING
		player.angle = 0
		player.collisionMode = 0
		player.value1 = 1
		CallFunction(Player_HandleRollAnimSpeed)
		player.state = Player_State_Air
		PlaySfx(SfxName[Jump], 0)
		player.collisionDisabled = true
		player.value30State = 1
	end if
end function

						end if
					end if

							player.animation = ANI_FLAILING2
						end if
					end if

public function Player_State_Air
	CallFunction(Player_HandleAirFriction)

	if player.gravity == GRAVITY_AIR
		CallFunction(Player_HandleAirMovement)

		if warp.destination > 0
			temp0 = player.yvel
			Abs(temp0)
			if temp0 < 0x60000
				temp0 = player.xvel
				Abs(temp0)
				if temp0 < 0x60000
					temp0 = player.xvel
					temp0 -= warp.speedCompare
					Abs(temp0)
					if temp0 > 0x40000
						if warp.value1 > 99
							if warp.value1 < 204
								warp.destination = 0
							end if
						end if
						warp.value1 = 0
					end if
				end if
			end if
			warp.speedCompare = player.xvel
		end if

		if player.yvel > 0x20000
			if player.animation == ANI_FLAILING1
				player.animation = ANI_WALKING
			end if

			if player.animation == ANI_FLAILING2
				player.animation = ANI_WALKING
			end if
		end if

		if player.animation == ANI_BOUNCING
			if player.yvel >= 0
				if player.animationReserve == ANI_STOPPED
					player.animationReserve = ANI_WALKING
				end if
				player.animation = player.animationReserve
			end if
		end if

		if player.animation == ANI_HURT
			if player.yvel >= 0
				if player.animationReserve == ANI_STOPPED
					player.animationReserve = ANI_WALKING
				end if
				player.animation = player.animationReserve
			end if
		end if

		if player.animation == ANI_JUMPING
			if player.value30State == 1
				if player.yvel >= player.value27
					CallFunction(player.value30)
				end if
			end if
		end if

		if player.animation == ANI_SKIDDING
			if player.skidding > 0
				player.skidding--
			else
				player.animation = ANI_WALKING
				player.prevAnimation = ANI_WALKING
				player.frame = 0
				player.animationSpeed = 40
			end if
		end if
	else
		player.state = Player_State_Ground
		CallFunction(Player_HandleOnGround)
		player.skidding = 0
	end if
end function

		end if
	end if
	temp1 = player.xpos
	temp1 >>= 16
	temp2 = player.ypos
	temp2 >>= 16
	temp2 += player.collisionTop
	Get16x16TileInfo(temp0, temp1, temp2, 8)

	end if

end function


public function Player_State_TubeRoll
	if player.gravity == GRAVITY_AIR
		player.state = Player_State_Air
		player.value1 = 0
		CallFunction(Player_HandleAirMovement)
	else
		if player.speed > 0
			if player.speed < object.value17
				player.speed = object.value17
			end if
			if player.speed > 1048576
				player.speed = 1048576
			end if
		else
			temp0 = object.value17
			FlipSign(temp0)
			if player.speed > temp0
				player.speed = temp0
				CallFunction(Player_HandleOnGround)
			end if
			if player.speed < -1048576
				player.speed = -1048576
			end if
		end if
		temp0 = player.speed
		Abs(temp0)
		if temp0 < 390594
			if warp.destination > 0
				if warp.value1 > 99
					if warp.value1 < 204
						warp.destination = 0
					end if
				end if
				warp.value1 = 0
			end if

	if stage.debugMode == true
		CallFunction(Player_ProcessUpdate)
		CheckEqual(options.attractMode, false)
		temp0 = checkResult
		CheckEqual(keyPress[0].buttonB, true)
		temp0 &= checkResult
		if temp0 == true
			player.type = TypeName[Debug Mode]

			player.state = Player_State_Static
			player.frame = 0
			player.rotation = 0
			player.interaction = false

			player.value8 = 0
			player.visible = 1
			player.value2 = 0
			object.value3 = 0
			object.value4 = 0

			screen[0].cameraStyle = options.originalControls
			object.value39 = C_BOX
			object.value40 = C_BOX
			object.value41 = C_BOX
			object.value42 = C_BOX

				stage.state = STAGE_RUNNING
			end if

			end if
			if object[+playerCount].propertyValue == 3
				object[+playerCount].type = invincibilityType
				object[+playerCount].propertyValue = 0
			end if
		else
			if player.gravity == GRAVITY_GROUND
				player.value30State = 0
			end if
			CallFunction(player.state)
			ProcessAnimation()
			if player.entityPos == screen[0].cameraTarget
				if player.animation == ANI_JUMPING
					screen[0].adjustCameraY = player.value29
				else
					if screen[0].adjustCameraY == player.value29
						screen[0].adjustCameraY = 0
						player.iypos += player.value29
					end if
				end if
			end if
			if player.collisionDisabled == 0
				temp0 = player.prevGravity
				player.prevGravity = player.gravity
				ProcessObjectMovement()
				player.prevGravity ^= GRAVITY_AIR
				CheckEqual(player.gravity, 0)
				player.prevGravity |= checkResult
				player.prevGravity ^= GRAVITY_AIR

				if temp0 == GRAVITY_AIR
					if player.prevGravity == GRAVITY_GROUND
					// in Sonic 1, achieveRingCount would also go here
						player.value18 = 0
						if player.animation == ANI_JUMPING
							if player.down == false
								if player.state != Player_State_BubbleBounce
									if player.state != Player_State_Air
										if player.state != Player_State_TubeRoll
											player.animation = ANI_WALKING
											if player.entityPos == screen[0].cameraTarget
												screen[0].adjustCameraY = 0
											end if
											player.iypos += player.value29
										end if
									end if
								end if
							end if
						end if
					end if
				end if
			else
				player.collisionDisabled = false
			end if
		end if
	else
		CallFunction(Player_ProcessUpdate)
		if player.gravity == GRAVITY_GROUND
			player.value30State = 0
		end if

		CallFunction(player.state)

		ProcessAnimation()
		if player.entityPos == screen[0].cameraTarget
			if player.animation == ANI_JUMPING
				screen[0].adjustCameraY = player.value29
			else
				if screen[0].adjustCameraY == player.value29
					screen[0].adjustCameraY = 0
					player.iypos += player.value29
				end if
			end if
		end if

		if player.collisionDisabled == false
				temp0 = player.prevGravity
				player.prevGravity = player.gravity
				ProcessObjectMovement()
				player.prevGravity ^= 1
				CheckEqual(player.gravity, 0)
				player.prevGravity |= checkResult
				player.prevGravity ^= 1
				if temp0 == 1
					if player.prevGravity == 0
						if player.animation == ANI_DROPDASH
							player.state = Player_State_Roll
							player.animation = 2
							PlaySfx(SfxName[Release], 0)
							if player.direction == 0
								player.speed = 786432
							else
								player.speed = -786432
							end if
							player.xvel = player.speed
						else
							if player.animation == 2
								if player.down == false
									player.animation = 1
									player.iypos += player.value29
								end if
							end if
						end if
					end if
				end if
			else
				player.collisionDisabled = false
			end if
	end if
end event

public function Player_Sidekick_UpdateAI
	if object.value14 == true
		object.up = object[0].up
		object.down = object[0].down
		object.left = object[0].left
		object.right = object[0].right
		object.jumpHold = object[0].jumpHold
		object.jumpPress = object[0].jumpPress
		return
	end if
	temp0 = saveRAM[1200]
	temp0 -= 32
	temp0 &= 63
	temp1 = saveRAM[1000 + temp0]
	temp2 = saveRAM[1100 + temp0]
	object.left = false
	object.right = false
	object.jumpPress = false
	if object.xpos < temp1
		object.right = true
	else
		if object.xpos > temp1
			object.left = true
		end if
	end if
	if object.gravity == 0
		if object.ypos > temp2
			object.jumpPress = true
		end if
	else
		if object.ypos > temp2
			object.jumpHold = true
		else
			object.jumpHold = false
		end if
	end if
end function
