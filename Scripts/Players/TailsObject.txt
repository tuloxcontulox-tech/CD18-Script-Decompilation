public function Player_ProcessUpdate
	if object.value15 == true
		CallFunction(Player_Sidekick_UpdateAI)
	else
		ProcessObjectControl()
		if playerCount > 1
			if object.state == Player_State_Air
				if object[1].state == Player_State_Fly
					if object[1].value14 == false
						temp0 = object.xpos
						temp0 -= object[1].xpos
						Abs(temp0)
						if temp0 < 1048576
							temp0 = object.ypos
							temp0 -= object[1].ypos
							Abs(temp0)
							if temp0 < 1572864
								object.state = Player_State_Carried
								object[1].value14 = true
							end if
						end if
					end if
				end if
			end if
		end if
		if object.entityPos == 0
			temp0 = saveRAM[1200]
			temp1 = 1000
			temp1 += temp0
			saveRAM[temp1] = object.xpos
			temp1 = 1100
			temp1 += temp0
			saveRAM[temp1] = object.ypos
			temp0++
			temp0 &= 63
			saveRAM[1200] = temp0
		end if
	end if
end function

public function Player_Action_DropDash
	if object.propertyValue < 2
		if object.animation == 2
			if object.jumpHold == true
				object.value2++
				if object.value2 >= 20
					object.animation = ANI_DROPDASH
					if object.value2 == 20
						PlaySfx(SfxName[Charge], 0)
					end if
				end if
			else
				object.value2 = 0
			end if
		end if
	end if
end function

public function Player_Sidekick_UpdateAI
	if object.value14 == true
		object.up = object[0].up
		object.down = object[0].down
		object.left = object[0].left
		object.right = object[0].right
		object.jumpHold = object[0].jumpHold
		object.jumpPress = object[0].jumpPress
		return
	end if
	temp0 = saveRAM[1200]
	temp0 -= 32
	temp0 &= 63
	temp1 = saveRAM[1000 + temp0]
	temp2 = saveRAM[1100 + temp0]
	object.left = false
	object.right = false
	object.jumpPress = false
	if object.xpos < temp1
		object.right = true
	else
		if object.xpos > temp1
			object.left = true
		end if
	end if
	if object.gravity == 0
		if object.ypos > temp2
			object.jumpPress = true
		end if
	else
		if object.ypos > temp2
			object.jumpHold = true
		else
			object.jumpHold = false
		end if
	end if
end function

public function Player_Action_Glide
	if object.jumpPress == true
		object.state = Player_State_Glide
		object.animation = ANI_GLIDING
		object.yvel = 32768
		if object.direction == 0
			object.speed = 262144
		else
			object.speed = -262144
		end if
		object.xvel = object.speed
	end if
end function

public function Player_State_Glide
	object.yvel = 32768
	object.xvel = object.speed
	if object.jumpHold == false
		object.state = Player_State_GlideDrop
		object.animation = ANI_GLIDE_DROP
	end if
	if object.direction == 0
		temp0 = 10
	else
		temp0 = -10
	end if
	ObjectTileCollision(2, temp0, 0, object.value33)
	if checkResult == true
		object.state = Player_State_Climb
		object.animation = ANI_CLIMBING
		object.speed = 0
		object.xvel = 0
	end if
	if object.gravity == 0
		object.state = Player_State_Ground
		object.animation = 0
	end if
end function

public function Player_State_GlideDrop
	object.yvel += object.value24
	object.xvel = 0
	if object.gravity == 0
		object.state = Player_State_Ground
		object.animation = 0
	end if
end function

public function Player_State_Climb
	object.xvel = 0
	object.yvel = 0
	if object.up == true
		object.ypos -= 65536
	else
		if object.down == true
			object.ypos += 65536
		end if
	end if
	if object.direction == 0
		temp0 = 10
	else
		temp0 = -10
	end if
	ObjectTileCollision(2, temp0, -16, object.value33)
	if checkResult == false
		object.state = Player_State_LedgePullUp
		object.animation = ANI_LEDGE_PULLUP
		object.value1 = 0
	end if
	if object.jumpPress == true
		object.state = Player_State_Air
		object.yvel = -262144
	end if
end function

public function Player_State_LedgePullUp
	object.value1++
	if object.value1 > 20
		object.state = Player_State_Ground
		object.ypos -= 1572864
		if object.direction == 0
			object.xpos += 1048576
		else
			object.xpos -= 1048576
		end if
	end if
end function

public function Player_State_Carried
	object.xpos = object[1].xpos
	object.ypos = object[1].ypos
	object.ypos += 1048576
	object.direction = object[1].direction
	if object.jumpPress == true
		object.state = Player_State_Air
		object.yvel = -262144
		object[1].value14 = false
	end if
	if object[1].state != Player_State_Fly
		object.state = Player_State_Air
		object[1].value14 = false
	end if
end function
