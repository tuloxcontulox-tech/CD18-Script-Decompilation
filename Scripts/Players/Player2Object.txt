// Physics Tables
private table Player_SonicPhysicsTable
	0x60000, 0x0C00, 0x1800, 0x0600, 0x08000, 0x0600, 0x68000, -0x40000
	0x30000, 0x0600, 0x0C00, 0x0300, 0x04000, 0x0300, 0x38000, -0x20000
end table

private table Player_TailsPhysicsTable
	0x60000, 0x0C00, 0x1800, 0x0600, 0x08000, 0x0600, 0x68000, -0x40000
	0x30000, 0x0600, 0x0C00, 0x0300, 0x04000, 0x0300, 0x38000, -0x20000
end table

private table Player_KnucklesPhysicsTable
	0x60000, 0x0C00, 0x1800, 0x0600, 0x08000, 0x0600, 0x60000, -0x40000
	0x30000, 0x0600, 0x0C00, 0x0300, 0x04000, 0x0300, 0x30000, -0x20000
end table

public function Player_UpdatePhysicsState
	switch player[currentPlayer].character
	case 0: temp0 = 0; break
	case 1: temp0 = 0; break
	case 2: temp0 = 1; break
	case 3: temp0 = 2; break
	end switch

	temp1 = 0
	if player[currentPlayer].ypos > stage.waterLevel
		temp1 = 8
		player[currentPlayer].gravityStrength = 0x1000
	else
		player[currentPlayer].gravityStrength = 0x3800
	end if

	switch temp0
	case 0: GetTableValue(player[currentPlayer].topSpeed, temp1, Player_SonicPhysicsTable); break
	case 1: GetTableValue(player[currentPlayer].topSpeed, temp1, Player_TailsPhysicsTable); break
	case 2: GetTableValue(player[currentPlayer].topSpeed, temp1, Player_KnucklesPhysicsTable); break
	end switch
	temp1++
	switch temp0
	case 0: GetTableValue(player[currentPlayer].acceleration, temp1, Player_SonicPhysicsTable); break
	case 1: GetTableValue(player[currentPlayer].acceleration, temp1, Player_TailsPhysicsTable); break
	case 2: GetTableValue(player[currentPlayer].acceleration, temp1, Player_KnucklesPhysicsTable); break
	end switch
	player[currentPlayer].deceleration = player[currentPlayer].acceleration
	temp1 += 5
	switch temp0
	case 0: GetTableValue(player[currentPlayer].jumpStrength, temp1, Player_SonicPhysicsTable); break
	case 1: GetTableValue(player[currentPlayer].jumpStrength, temp1, Player_TailsPhysicsTable); break
	case 2: GetTableValue(player[currentPlayer].jumpStrength, temp1, Player_KnucklesPhysicsTable); break
	end switch
	temp1++
	switch temp0
	case 0: GetTableValue(player[currentPlayer].jumpCap, temp1, Player_SonicPhysicsTable); break
	case 1: GetTableValue(player[currentPlayer].jumpCap, temp1, Player_TailsPhysicsTable); break
	case 2: GetTableValue(player[currentPlayer].jumpCap, temp1, Player_KnucklesPhysicsTable); break
	end switch
end function

public function Player_ProcessUpdate
	if player.isSidekick == true
		CallFunction(Player_Sidekick_UpdateAI)
	else
		ProcessObjectControl()
		if playerCount > 1
			if player.state == Player_State_Air
				if object[1].state == Player_State_Fly
					if object[1].carrying == false
						temp0 = player.xpos - object[1].xpos
						Abs(temp0)
						if temp0 < 0x100000
							temp0 = player.ypos - object[1].ypos
							Abs(temp0)
							if temp0 < 0x180000
								player.state = Player_State_Carried
								object[1].carrying = true
							end if
						end if
					end if
				end if
			end if
		end if
		if player.entityPos == 0
			temp0 = saveRAM[1200]
			saveRAM[1000 + temp0] = player.xpos
			saveRAM[1100 + temp0] = player.ypos
			saveRAM[1200] = (temp0 + 1) & 63
		end if
	end if
end function


public function Player_Action_DropDash
	if player.character < 2 // Sonic
		if player.animation == 2 // ANI_JUMPING
			if player.jumpHold == true
				player.abilityTimer++
				if player.abilityTimer >= 20
					player.animation = ANI_DROPDASH
					if player.abilityTimer == 20: PlaySfx(SfxName[Charge], 0)
				end if
			else
				player.abilityTimer = 0
			end if
		end if
	end if
end function

public function Player_Sidekick_UpdateAI
	if player.carrying == true
		player.up = object[0].up; player.down = object[0].down
		player.left = object[0].left; player.right = object[0].right
		player.jumpHold = object[0].jumpHold; player.jumpPress = object[0].jumpPress
		return
	end if
	temp0 = (saveRAM[1200] - 32) & 63
	temp1 = saveRAM[1000 + temp0]
	temp2 = saveRAM[1100 + temp0]
	player.left = false; player.right = false; player.jumpPress = false
	if player.xpos < temp1: player.right = true
	elif player.xpos > temp1: player.left = true
	if player.gravity == 0
		if player.ypos > temp2: player.jumpPress = true
	else: player.jumpHold = (player.ypos > temp2)
end function

public function Player_Action_Glide
	if player.jumpPress == true
		player.state = Player_State_Glide
		player.animation = ANI_GLIDING
		player.yvel = 0x8000
		player.speed = 0x40000 if player.direction == 0 else -0x40000
		player.xvel = player.speed
	end if
end function

public function Player_State_Glide
	player.yvel = 0x8000; player.xvel = player.speed
	if player.jumpHold == false
		player.state = Player_State_GlideDrop
		player.animation = ANI_GLIDE_DROP
	end if
	temp0 = 10 if player.direction == 0 else -10
	ObjectTileCollision(2, temp0, 0, player.collisionPlane)
	if checkResult == true: player.state = Player_State_Climb; player.animation = ANI_CLIMBING; player.speed = 0; player.xvel = 0
	if player.gravity == 0: player.state = Player_State_Ground; player.animation = 0
end function

public function Player_State_GlideDrop
	player.yvel += player.gravityStrength; player.xvel = 0
	if player.gravity == 0: player.state = Player_State_Ground; player.animation = 0
end function

public function Player_State_Climb
	player.xvel = 0; player.yvel = 0
	if player.up == true: player.ypos -= 0x10000
	elif player.down == true: player.ypos += 0x10000
	temp0 = 10 if player.direction == 0 else -10
	ObjectTileCollision(2, temp0, -16, player.collisionPlane)
	if checkResult == false: player.state = Player_State_LedgePullUp; player.animation = ANI_LEDGE_PULLUP; player.timer = 0
	if player.jumpPress == true: player.state = Player_State_Air; player.yvel = -0x40000
end function

public function Player_State_LedgePullUp
	player.timer++
	if player.timer > 20: player.state = Player_State_Ground; player.ypos -= 0x180000; player.xpos += (0x100000 if player.direction == 0 else -0x100000)
end function

public function Player_State_Carried
	player.xpos = object[1].xpos; player.ypos = object[1].ypos + 0x100000
	player.direction = object[1].direction
	if player.jumpPress == true: player.state = Player_State_Air; player.yvel = -0x40000; object[1].carrying = false
	if object[1].state != Player_State_Fly: player.state = Player_State_Air; object[1].carrying = false
end function
