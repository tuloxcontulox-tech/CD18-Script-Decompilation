event ObjectUpdate
	if object.state == 0
		if options.gameMode == 0
			CallFunction(Start_HandleCheatCodes)
		end if
	end if
	switch object.state
	case START_SETUP
		//This still has touchscreen support?
		CheckTouchRect(0, 0, screen.xsize, 240)
		if keyPress[0].start == true
			checkResult = false
		end if
		if keyPress[0].buttonA == true
			checkResult = false
		end if
		if keyPress[0].buttonB == true
			checkResult = false
		end if
		if keyPress[0].buttonC == true
			checkResult = false
		end if
		if checkResult > -1
			object.state = MENU_MAIN
			object.value1 = 0
			object.value4 = 0
			object.value5 = 0
			LoadSpriteSheet("Title/Menu.gif")
			PlaySfx(SfxName[Select], 0)
		else
			object.value4++
			if object.value4 == 1550
				object.state = 5
			end if
		end if
		break
	case MENU_MAIN
		if object.value5 > 0
			object.value5--
		else
			if keyPress[0].left == true
				if object.value1 > 0
					object.value1--
					object.value4 = 1 // moving left
					object.value5 = 15 // 0.25 second transition (15 frames)
					PlaySfx(SfxName[Menu Button], 0)
				end if
			end if
			if keyPress[0].right == true
				if object.value1 < 4
					object.value1++
					object.value4 = 2 // moving right
					object.value5 = 15 // 0.25 second transition (15 frames)
					PlaySfx(SfxName[Menu Button], 0)
				end if
			end if
			if keyPress[0].buttonB == true
				PlaySfx(SfxName[Back Button], 0)
				object.state = START_SETUP
				LoadSpriteSheet("Title/Title.gif")
			end if
			checkResult = false
			if keyPress[0].start == true
				checkResult = true
			end if
			if keyPress[0].buttonA == true
				checkResult = true
			end if
			if keyPress[0].buttonC == true
				checkResult = true
			end if
			if checkResult == true
				PlaySfx(SfxName[Select], 0)
				switch object.value1
				case 0
					object.state = START_NEWGAME
					StopMusic()
					break
				case 1
					stage.activeList = PRESENTATION_STAGE
					stage.listPos = 2
					LoadStage()
					break
				case 2
					object.state = MENU_SOUNDTRACK
					object.value2 = 0
					break
				case 3
					object.state = MENU_EXTRAS
					object.value2 = 0
					break
				case 4
					object.state = MENU_EXIT
					screen.fade = 0
					break
				end switch
			end if
		end if
		break
	case MENU_EXIT
		if keyPress[0].buttonB == true
			object.state = MENU_MAIN
			screen.fade = 0
			SetScreenFade(0, 0, 0, 0)
			PlaySfx(SfxName[Back Button], 0)
		else
			if screen.fade < 255
				screen.fade += 4
				if screen.fade > 255
					screen.fade = 255
				end if
				SetScreenFade(0, 0, 0, screen.fade)
			else
				SetScreenFade(0, 0, 0, 255)
				engine.state = 8
			end if
		end if
		break

	case MENU_SOUNDTRACK
		if keyPress[0].left == true
			if object.value2 != 0
				object.value2 = 0
				PlaySfx(SfxName[Menu Button], 0)
			end if
		end if
		if keyPress[0].right == true
			if object.value2 != 1
				object.value2 = 1
				PlaySfx(SfxName[Menu Button], 0)
			end if
		end if
		if keyPress[0].up == true
			if object.value2 == 2
				object.value2 = options.soundtrack
				PlaySfx(SfxName[Menu Button], 0)
			end if
		end if
		if keyPress[0].down == true
			if object.value2 != 2
				object.value2 = 2
				PlaySfx(SfxName[Menu Button], 0)
			end if
		end if
		if keyPress[0].buttonB == true
			PlaySfx(SfxName[Back Button], 0)
			object.state = MENU_MAIN
		end if
		checkResult = false
		if keyPress[0].start == true
			checkResult = true
		end if
		if keyPress[0].buttonA == true
			checkResult = true
		end if
		if keyPress[0].buttonC == true
			checkResult = true
		end if
		if checkResult == true
			if object.value2 == 0
				options.soundtrack = 0
				PlaySfx(SfxName[Select], 0)
				PlayMusic(options.soundtrack)
			else
				if object.value2 == 1
					options.soundtrack = 1
					PlaySfx(SfxName[Select], 0)
					PlayMusic(options.soundtrack)
				else
					PlaySfx(SfxName[Back Button], 0)
					object.state = MENU_MAIN
				end if
			end if
		end if
		break
	case MENU_EXTRAS
		if keyPress[0].up == true
			object.value2--
			if object.value2 < 0
				object.value2 = 4
			end if
			PlaySfx(SfxName[Menu Button], 0)
		end if
		if keyPress[0].down == true
			object.value2++
			if object.value2 > 4
				object.value2 = 0
			end if
			PlaySfx(SfxName[Menu Button], 0)
		end if
		if keyPress[0].buttonB == true
			PlaySfx(SfxName[Back Button], 0)
			object.state = MENU_MAIN
		end if
		checkResult = false
		if keyPress[0].start == true
			checkResult = true
		end if
		if keyPress[0].buttonA == true
			checkResult = true
		end if
		if keyPress[0].buttonC == true
			checkResult = true
		end if
		if checkResult == true
			PlaySfx(SfxName[Select], 0)
			switch object.value2
			case 0
				// D.A. GARDEN
				stage.activeList = PRESENTATION_STAGE
				stage.listPos = 5
				LoadStage()
				break
			case 1
				// MODO VISUAL
				object.state = MENU_VISUALMODE
				object.value3 = 0
				break
			case 2
				// PRUEBA DE SONIDO
				stage.activeList = PRESENTATION_STAGE
				stage.listPos = 4
				LoadStage()
				break
			case 3
				// SELECCIONAR FASE
				stage.activeList = PRESENTATION_STAGE
				stage.listPos = 3
				LoadStage()
				break
			case 4
				// ATRÁS
				PlaySfx(SfxName[Back Button], 0)
				object.state = MENU_MAIN
				break
			end switch
		end if
		break
	case MENU_VISUALMODE
		if keyPress[0].up == true
			if object.value3 == 2
				object.value3 = 0
				PlaySfx(SfxName[Menu Button], 0)
			else
				if object.value3 == 3
					object.value3 = 1
					PlaySfx(SfxName[Menu Button], 0)
				else
					if object.value3 == 4
						object.value3 = 2
						PlaySfx(SfxName[Menu Button], 0)
					end if
				end if
			end if
		end if
		if keyPress[0].down == true
			if object.value3 == 0
				object.value3 = 2
				PlaySfx(SfxName[Menu Button], 0)
			else
				if object.value3 == 1
					object.value3 = 3
					PlaySfx(SfxName[Menu Button], 0)
				else
					if object.value3 < 4
						object.value3 = 4
						PlaySfx(SfxName[Menu Button], 0)
					end if
				end if
			end if
		end if
		if keyPress[0].left == true
			if object.value3 == 1
				object.value3 = 0
				PlaySfx(SfxName[Menu Button], 0)
			else
				if object.value3 == 3
					object.value3 = 2
					PlaySfx(SfxName[Menu Button], 0)
				end if
			end if
		end if
		if keyPress[0].right == true
			if object.value3 == 0
				object.value3 = 1
				PlaySfx(SfxName[Menu Button], 0)
			else
				if object.value3 == 2
					object.value3 = 3
					PlaySfx(SfxName[Menu Button], 0)
				end if
			end if
		end if
		if keyPress[0].buttonB == true
			PlaySfx(SfxName[Back Button], 0)
			object.state = MENU_EXTRAS
		end if
		checkResult = false
		if keyPress[0].start == true
			checkResult = true
		end if
		if keyPress[0].buttonA == true
			checkResult = true
		end if
		if keyPress[0].buttonC == true
			checkResult = true
		end if
		if checkResult == true
			if object.value3 == 0
				PlaySfx(SfxName[Select], 0)
				if options.soundtrack == 0
					CallNativeFunction2(PlayVideo, "Opening", 0)
				else
					CallNativeFunction2(PlayVideo, "OpeningUS", 0)
				end if
			else
				if object.value3 == 1
					PlaySfx(SfxName[Select], 0)
					CallNativeFunction2(PlayVideo, "PencilTest", 0)
				else
					if object.value3 == 2
						PlaySfx(SfxName[Select], 0)
						if options.soundtrack == 0
							CallNativeFunction2(PlayVideo, "GoodEnding", 0)
						else
							CallNativeFunction2(PlayVideo, "GoodEndingUS", 0)
						end if
					else
						if object.value3 == 3
							PlaySfx(SfxName[Select], 0)
							if options.soundtrack == 0
								CallNativeFunction2(PlayVideo, "BadEnding", 0)
							else
								CallNativeFunction2(PlayVideo, "BadEndingUS", 0)
							end if
						else
							PlaySfx(SfxName[Back Button], 0)
							object.state = MENU_EXTRAS
						end if
					end if
				end if
			end if
		end if
		break
	case START_NEWGAME
		screen.fade += 4
		if screen.fade == 384
		
			if options.soundtrack == 0
			// Interestingly, video playback was turned into a Native Function for this port.
				CallNativeFunction2(PlayVideo, "Opening", 0)
			else
				CallNativeFunction2(PlayVideo, "OpeningUS", 0)
			end if

			// Off to Palmtree Panic you go!
			stage.activeList 	= REGULAR_STAGE
			stage.listPos 		= 0
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_STAGESELECT
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 3
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_SOUNDTEST
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 4
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_DAGARDEN
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 5
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_DEMO
		screen.fade += 4
		if screen.fade == 384
			fadeColor = 0
			player.score = 0
			player.lives = 3
			player.scoreBonus 		= 50000
			generator.destroyed 	= false
			metalSonic.destroyed 	= false
			goodFuture.enabled 		= false
			goodFuture.list 		= 0
			goodFuture.count 		= 0
			options.gameMode 		= 0
			options.attractMode 	= true
			
			switch options.demoNumber
			case DEMO_R11A
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 0
				break
			case DEMO_SS1
				stage.activeList 	= SPECIAL_STAGE
				stage.listPos 		= 0
				break
			case DEMO_R43C
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 28
				break
			case DEMO_SS6
				stage.activeList 	= SPECIAL_STAGE
				stage.listPos = 5
				break
			case DEMO_R82A
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 64
				break
			end switch
			options.demoNumber++
			options.demoNumber %= 5
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	end switch
end event


event ObjectDraw
	switch object.state
	case START_SETUP
		object.animationTimer++
		if object.animationTimer > 59
			object.animationTimer = 0
		end if
		if object.animationTimer > 29
			if options.touchControls == false
				DrawSprite(0)
			else
				DrawSprite(1)
			end if
		end if
		break
	case MENU_MAIN
		temp0 = screen.xcenter

		// Arrow bobbing animation calculation
		object.value6++
		if object.value6 >= 60
			object.value6 = 0
		end if

		if object.value6 < 30
			temp6 = 0
		else
			temp6 = 2
		end if

		temp1 = temp0
		temp1 -= 72
		temp1 -= temp6 // Left arrow bobbing

		temp2 = temp0
		temp2 += 72
		temp2 += temp6 // Right arrow bobbing

		if object.value5 > 0
			// Slide animation active: object.value5 counts down from 15 to 0
			// Scale factor: object.value5 * 18 (so 15 * 18 = 270 offset, fully offscreen)
			temp7 = object.value5
			temp7 *= 18

			if object.value4 == 1
				// Moving left (to previous option)
				// Current option slides out to the right: temp0 + (270 - temp7)
				temp3 = 270
				temp3 -= temp7
				temp4 = temp0
				temp4 += temp3

				// Incoming option slides in from the left: temp0 - temp7
				temp5 = temp0
				temp5 -= temp7
			else
				// Moving right (to next option)
				// Current option slides out to the left: temp0 - (270 - temp7)
				temp3 = 270
				temp3 -= temp7
				temp4 = temp0
				temp4 -= temp3

				// Incoming option slides in from the right: temp0 + temp7
				temp5 = temp0
				temp5 += temp7
			end if

			// Draw incoming target option (object.value1) at temp5, and outgoing option at temp4
			// User requested ALL main menu options to always use yellow sprites (12, 4, 6, 10, 8)
			switch object.value1
			case 0
				DrawSpriteScreenXY(12, temp5, 205) // EMPEZAR PARTIDA (yellow) incoming
				DrawSpriteScreenXY(4, temp4, 205)  // CONTRARRELOJ (yellow) outgoing
				break
			case 1
				DrawSpriteScreenXY(4, temp5, 205)  // CONTRARRELOJ (yellow) incoming
				if object.value4 == 1
					DrawSpriteScreenXY(6, temp4, 205)  // BANDAS SONORAS (yellow) outgoing
				else
					DrawSpriteScreenXY(12, temp4, 205) // EMPEZAR PARTIDA (yellow) outgoing
				end if
				break
			case 2
				DrawSpriteScreenXY(6, temp5, 205)  // BANDAS SONORAS (yellow) incoming
				if object.value4 == 1
					DrawSpriteScreenXY(10, temp4, 205) // EXTRAS (yellow) outgoing
				else
					DrawSpriteScreenXY(4, temp4, 205)  // CONTRARRELOJ (yellow) outgoing
				end if
				break
			case 3
				DrawSpriteScreenXY(10, temp5, 205) // EXTRAS (yellow) incoming
				if object.value4 == 1
					DrawSpriteScreenXY(8, temp4, 205)  // SALIR DEL JUEGO (yellow) outgoing
				else
					DrawSpriteScreenXY(6, temp4, 205)  // BANDAS SONORAS (yellow) outgoing
				end if
				break
			case 4
				DrawSpriteScreenXY(8, temp5, 205)  // SALIR DEL JUEGO (yellow) incoming
				DrawSpriteScreenXY(10, temp4, 205) // EXTRAS (yellow) outgoing
				break
			end switch
		else
			// Static option (All yellow sprites: 12, 4, 6, 10, 8)
			if object.value1 == 0
				DrawSpriteScreenXY(12, temp0, 205)
				DrawSpriteScreenXY(14, temp2, 205)
			else
				if object.value1 == 1
					DrawSpriteScreenXY(4, temp0, 205)
					DrawSpriteScreenXY(13, temp1, 205)
					DrawSpriteScreenXY(14, temp2, 205)
				else
					if object.value1 == 2
						DrawSpriteScreenXY(6, temp0, 205)
						DrawSpriteScreenXY(13, temp1, 205)
						DrawSpriteScreenXY(14, temp2, 205)
					else
						if object.value1 == 3
							DrawSpriteScreenXY(10, temp0, 205)
							DrawSpriteScreenXY(13, temp1, 205)
							DrawSpriteScreenXY(14, temp2, 205)
						else
							DrawSpriteScreenXY(8, temp0, 205)
							DrawSpriteScreenXY(13, temp1, 205)
						end if
					end if
				end if
			end if
		end if
		break
	case MENU_EXIT
		temp0 = screen.xcenter
		temp1 = temp0
		temp1 -= 72
		DrawSpriteScreenXY(8, temp0, 205)
		DrawSpriteScreenXY(13, temp1, 205)
		break
	case MENU_SOUNDTRACK
		temp0 = screen.xcenter
		temp1 = temp0
		temp1 -= 110
		temp2 = temp0
		temp2 -= 45
		temp3 = temp0
		temp3 += 45
		temp4 = temp0
		temp4 += 110

		// Draw 256x160 panel background
		DrawSpriteScreenXY(44, temp0, 120)

		// Draw Header BANDAS SONORAS at bottom left
		DrawSpriteScreenXY(24, temp1, 215)

		// Eggman DJ 2-frame animation (alternates every 32 frames)
		object.value7++
		if object.value7 >= 64
			object.value7 = 0
		end if
		if object.value7 < 32
			DrawSpriteScreenXY(25, temp0, 72) // Eggman DJ Frame 0
		else
			DrawSpriteScreenXY(46, temp0, 72) // Eggman DJ Frame 1
		end if

		// Music notes sine oscillation next to Eggman
		object.value8 += 4
