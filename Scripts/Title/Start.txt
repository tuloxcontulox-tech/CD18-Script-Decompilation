event ObjectUpdate
	if object.state == 0
		if options.gameMode == 0
			CallFunction(Start_HandleCheatCodes)
		end if
	end if
	switch object.state
	case START_SETUP
		//This still has touchscreen support?
		CheckTouchRect(0, 0, screen.xsize, 240)
		if keyPress[0].start == true
			checkResult = false
		end if
		if keyPress[0].buttonA == true
			checkResult = false
		end if
		if keyPress[0].buttonB == true
			checkResult = false
		end if
		if keyPress[0].buttonC == true
			checkResult = false
		end if
		if checkResult > -1
			object.state = MENU_MAIN
			object.value1 = 0
			PlaySfx(SfxName[Select], 0)
		else
			object.value4++
			if object.value4 == 1550
				object.state = 5
			end if
		end if
		break
	case MENU_MAIN
		if keyPress[0].up == true
			object.value1--
			if object.value1 < 0
				object.value1 = 4
			end if
			PlaySfx(SfxName[Menu Button], 0)
		end if
		if keyPress[0].down == true
			object.value1++
			if object.value1 > 4
				object.value1 = 0
			end if
			PlaySfx(SfxName[Menu Button], 0)
		end if
		if keyPress[0].left == true
			object.value1--
			if object.value1 < 0
				object.value1 = 4
			end if
			PlaySfx(SfxName[Menu Button], 0)
		end if
		if keyPress[0].right == true
			object.value1++
			if object.value1 > 4
				object.value1 = 0
			end if
			PlaySfx(SfxName[Menu Button], 0)
		end if
		if keyPress[0].buttonB == true
			PlaySfx(SfxName[Back Button], 0)
			object.state = START_SETUP
		end if
		checkResult = false
		if keyPress[0].start == true
			checkResult = true
		end if
		if keyPress[0].buttonA == true
			checkResult = true
		end if
		if keyPress[0].buttonC == true
			checkResult = true
		end if
		if checkResult == true
			PlaySfx(SfxName[Select], 0)
			switch object.value1
			case 0
				object.state = START_NEWGAME
				StopMusic()
				break
			case 1
				stage.activeList = PRESENTATION_STAGE
				stage.listPos = 2
				LoadStage()
				break
			case 2
				object.state = MENU_SOUNDTRACK
				object.value2 = 0
				break
			case 3
				object.state = MENU_EXTRAS
				object.value2 = 0
				break
			case 4
				engine.state = 8
				break
			end switch
		end if
		break
	case MENU_SOUNDTRACK
		if keyPress[0].left == true
			if object.value2 != 0
				object.value2 = 0
				PlaySfx(SfxName[Menu Button], 0)
			end if
		end if
		if keyPress[0].right == true
			if object.value2 != 1
				object.value2 = 1
				PlaySfx(SfxName[Menu Button], 0)
			end if
		end if
		if keyPress[0].up == true
			if object.value2 == 2
				object.value2 = options.soundtrack
				PlaySfx(SfxName[Menu Button], 0)
			end if
		end if
		if keyPress[0].down == true
			if object.value2 != 2
				object.value2 = 2
				PlaySfx(SfxName[Menu Button], 0)
			end if
		end if
		if keyPress[0].buttonB == true
			PlaySfx(SfxName[Back Button], 0)
			object.state = MENU_MAIN
		end if
		checkResult = false
		if keyPress[0].start == true
			checkResult = true
		end if
		if keyPress[0].buttonA == true
			checkResult = true
		end if
		if keyPress[0].buttonC == true
			checkResult = true
		end if
		if checkResult == true
			if object.value2 == 0
				options.soundtrack = 0
				PlaySfx(SfxName[Select], 0)
				PlayMusic(options.soundtrack)
			else
				if object.value2 == 1
					options.soundtrack = 1
					PlaySfx(SfxName[Select], 0)
					PlayMusic(options.soundtrack)
				else
					PlaySfx(SfxName[Back Button], 0)
					object.state = MENU_MAIN
				end if
			end if
		end if
		break
	case MENU_EXTRAS
		if keyPress[0].up == true
			object.value2--
			if object.value2 < 0
				object.value2 = 4
			end if
			PlaySfx(SfxName[Menu Button], 0)
		end if
		if keyPress[0].down == true
			object.value2++
			if object.value2 > 4
				object.value2 = 0
			end if
			PlaySfx(SfxName[Menu Button], 0)
		end if
		if keyPress[0].buttonB == true
			PlaySfx(SfxName[Back Button], 0)
			object.state = MENU_MAIN
		end if
		checkResult = false
		if keyPress[0].start == true
			checkResult = true
		end if
		if keyPress[0].buttonA == true
			checkResult = true
		end if
		if keyPress[0].buttonC == true
			checkResult = true
		end if
		if checkResult == true
			PlaySfx(SfxName[Select], 0)
			switch object.value2
			case 0
				// D.A. GARDEN
				stage.activeList = PRESENTATION_STAGE
				stage.listPos = 5
				LoadStage()
				break
			case 1
				// MODO VISUAL
				object.state = MENU_VISUALMODE
				object.value3 = 0
				break
			case 2
				// PRUEBA DE SONIDO
				stage.activeList = PRESENTATION_STAGE
				stage.listPos = 4
				LoadStage()
				break
			case 3
				// SELECCIONAR FASE
				stage.activeList = PRESENTATION_STAGE
				stage.listPos = 3
				LoadStage()
				break
			case 4
				// ATRÁS
				PlaySfx(SfxName[Back Button], 0)
				object.state = MENU_MAIN
				break
			end switch
		end if
		break
	case MENU_VISUALMODE
		if keyPress[0].up == true
			if object.value3 == 2
				object.value3 = 0
				PlaySfx(SfxName[Menu Button], 0)
			else
				if object.value3 == 3
					object.value3 = 1
					PlaySfx(SfxName[Menu Button], 0)
				else
					if object.value3 == 4
						object.value3 = 2
						PlaySfx(SfxName[Menu Button], 0)
					end if
				end if
			end if
		end if
		if keyPress[0].down == true
			if object.value3 == 0
				object.value3 = 2
				PlaySfx(SfxName[Menu Button], 0)
			else
				if object.value3 == 1
					object.value3 = 3
					PlaySfx(SfxName[Menu Button], 0)
				else
					if object.value3 < 4
						object.value3 = 4
						PlaySfx(SfxName[Menu Button], 0)
					end if
				end if
			end if
		end if
		if keyPress[0].left == true
			if object.value3 == 1
				object.value3 = 0
				PlaySfx(SfxName[Menu Button], 0)
			else
				if object.value3 == 3
					object.value3 = 2
					PlaySfx(SfxName[Menu Button], 0)
				end if
			end if
		end if
		if keyPress[0].right == true
			if object.value3 == 0
				object.value3 = 1
				PlaySfx(SfxName[Menu Button], 0)
			else
				if object.value3 == 2
					object.value3 = 3
					PlaySfx(SfxName[Menu Button], 0)
				end if
			end if
		end if
		if keyPress[0].buttonB == true
			PlaySfx(SfxName[Back Button], 0)
			object.state = MENU_EXTRAS
		end if
		checkResult = false
		if keyPress[0].start == true
			checkResult = true
		end if
		if keyPress[0].buttonA == true
			checkResult = true
		end if
		if keyPress[0].buttonC == true
			checkResult = true
		end if
		if checkResult == true
			if object.value3 == 0
				PlaySfx(SfxName[Select], 0)
				if options.soundtrack == 0
					CallNativeFunction2(PlayVideo, "Opening", 0)
				else
					CallNativeFunction2(PlayVideo, "OpeningUS", 0)
				end if
			else
				if object.value3 == 1
					PlaySfx(SfxName[Select], 0)
					CallNativeFunction2(PlayVideo, "PencilTest", 0)
				else
					if object.value3 == 2
						PlaySfx(SfxName[Select], 0)
						if options.soundtrack == 0
							CallNativeFunction2(PlayVideo, "GoodEnding", 0)
						else
							CallNativeFunction2(PlayVideo, "GoodEndingUS", 0)
						end if
					else
						if object.value3 == 3
							PlaySfx(SfxName[Select], 0)
							if options.soundtrack == 0
								CallNativeFunction2(PlayVideo, "BadEnding", 0)
							else
								CallNativeFunction2(PlayVideo, "BadEndingUS", 0)
							end if
						else
							PlaySfx(SfxName[Back Button], 0)
							object.state = MENU_EXTRAS
						end if
					end if
				end if
			end if
		end if
		break
	case START_NEWGAME
		screen.fade += 4
		if screen.fade == 384
		
			if options.soundtrack == 0
			// Interestingly, video playback was turned into a Native Function for this port.
				CallNativeFunction2(PlayVideo, "Opening", 0)
			else
				CallNativeFunction2(PlayVideo, "OpeningUS", 0)
			end if

			// Off to Palmtree Panic you go!
			stage.activeList 	= REGULAR_STAGE
			stage.listPos 		= 0
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_STAGESELECT
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 3
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_SOUNDTEST
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 4
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_DAGARDEN
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 5
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_DEMO
		screen.fade += 4
		if screen.fade == 384
			fadeColor = 0
			player.score = 0
			player.lives = 3
			player.scoreBonus 		= 50000
			generator.destroyed 	= false
			metalSonic.destroyed 	= false
			goodFuture.enabled 		= false
			goodFuture.list 		= 0
			goodFuture.count 		= 0
			options.gameMode 		= 0
			options.attractMode 	= true
			
			switch options.demoNumber
			case DEMO_R11A
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 0
				break
			case DEMO_SS1
				stage.activeList 	= SPECIAL_STAGE
				stage.listPos 		= 0
				break
			case DEMO_R43C
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 28
				break
			case DEMO_SS6
				stage.activeList 	= SPECIAL_STAGE
				stage.listPos = 5
				break
			case DEMO_R82A
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 64
				break
			end switch
			options.demoNumber++
			options.demoNumber %= 5
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	end switch
end event


event ObjectDraw
	switch object.state
	case START_SETUP
		object.animationTimer++
		if object.animationTimer > 59
			object.animationTimer = 0
		end if
		if object.animationTimer > 29
			if options.touchControls == false
				DrawSprite(0)
			else
				DrawSprite(1)
			end if
		end if
		break
	case MENU_MAIN
		temp0 = screen.xcenter
		// object.value1: 0 = EMPEZAR PARTIDA, 1 = CONTRARRELOJ, 2 = BANDAS SONORAS, 3 = EXTRAS, 4 = SALIR DEL JUEGO
		temp1 = temp0
		temp1 -= 72
		temp2 = temp0
		temp2 += 72
		if object.value1 == 0
			DrawSpriteScreenXY(12, temp0, 205)
			DrawSpriteScreenXY(13, temp1, 205)
			DrawSpriteScreenXY(14, temp2, 205)
		else
			if object.value1 == 1
				DrawSpriteScreenXY(4, temp0, 205)
				DrawSpriteScreenXY(13, temp1, 205)
				DrawSpriteScreenXY(14, temp2, 205)
			else
				if object.value1 == 2
					DrawSpriteScreenXY(6, temp0, 205)
					DrawSpriteScreenXY(13, temp1, 205)
					DrawSpriteScreenXY(14, temp2, 205)
				else
					if object.value1 == 3
						DrawSpriteScreenXY(10, temp0, 205)
						DrawSpriteScreenXY(13, temp1, 205)
						DrawSpriteScreenXY(14, temp2, 205)
					else
						DrawSpriteScreenXY(8, temp0, 205)
						DrawSpriteScreenXY(13, temp1, 205)
						DrawSpriteScreenXY(14, temp2, 205)
					end if
				end if
			end if
		end if
		break
	case MENU_SOUNDTRACK
		temp0 = screen.xcenter
		temp1 = temp0
		temp1 -= 40
		temp2 = temp0
		temp2 -= 45
		temp3 = temp0
		temp3 += 45
		temp4 = temp0
		temp4 += 80
		// Draw Header BANDAS SONORAS
		DrawSpriteScreenXY(24, temp1, 60)
		// Draw Eggman DJ
		DrawSpriteScreenXY(25, temp0, 110)
		// Draw JP Button
		if options.soundtrack == 0
			DrawSpriteScreenXY(26, temp2, 160)
		else
			DrawSpriteScreenXY(27, temp2, 160)
		end if
		// Draw US Button
		if options.soundtrack == 1
			DrawSpriteScreenXY(28, temp3, 160)
		else
			DrawSpriteScreenXY(29, temp3, 160)
		end if
		// Draw ATRÁS Button
		if object.value2 == 2
			DrawSpriteScreenXY(31, temp4, 215)
		else
			DrawSpriteScreenXY(30, temp4, 215)
		end if
		break
	case MENU_EXTRAS
		temp0 = screen.xcenter
		temp1 = temp0
		temp1 -= 40
		temp2 = temp0
		temp2 += 80
		// Draw Header EXTRAS
		DrawSpriteScreenXY(23, temp1, 45)
		// D. A. GARDEN
		if object.value2 == 0
			DrawSpriteScreenXY(16, temp0, 85)
		else
			DrawSpriteScreenXY(15, temp0, 85)
		end if
		// MODO VISUAL
		if object.value2 == 1
			DrawSpriteScreenXY(18, temp0, 115)
		else
			DrawSpriteScreenXY(17, temp0, 115)
		end if
		// PRUEBA DE SONIDO
		if object.value2 == 2
			DrawSpriteScreenXY(20, temp0, 145)
		else
			DrawSpriteScreenXY(19, temp0, 145)
		end if
		// SELECCIONAR FASE
		if object.value2 == 3
			DrawSpriteScreenXY(22, temp0, 175)
		else
			DrawSpriteScreenXY(21, temp0, 175)
		end if
		// ATRÁS
		if object.value2 == 4
			DrawSpriteScreenXY(31, temp2, 215)
		else
			DrawSpriteScreenXY(30, temp2, 215)
		end if
		break
	case MENU_VISUALMODE
		temp0 = screen.xcenter
		temp1 = temp0
		temp1 -= 40
		temp2 = temp0
		temp2 -= 60
		temp3 = temp0
		temp3 += 60
		temp4 = temp0
		temp4 += 80
		// Header EXTRAS
		DrawSpriteScreenXY(23, temp1, 40)

		// Thumbnail 0: COMIENZO
		DrawSpriteScreenXY(32, temp2, 90)

event ObjectStartup
	LoadSpriteSheet("Title/Title.gif")
	switch engine.language
	case LANGUAGE_EN
	case LANGUAGE_JP
	case LANGUAGE_PT
	case LANGUAGE_RU
		SpriteFrame(-43, 0, 87, 11, 106, 103)
		SpriteFrame(-56, 0, 117, 11, 106, 91)
		break

	case LANGUAGE_FR
		SpriteFrame(-60, 0, 120, 11, 391, 387)
		SpriteFrame(-94, 0, 189, 11, 322, 399)
